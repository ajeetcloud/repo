
List<String> words;

 public List<String> getWords() {
        words = new ArrayList<>();
        TrieNode node = root;
        getListDFS(node, "");

}

private void getListDFS(TrieNode node, String word) {
    if (node == null) {
        return null;
    }
    if (node.isEnd()) {
        words.add(word);
    }

    for (int i = 0; i < 26; i++) {
        TrieNode child = node.children[i];
        if (child != null) {
            word += (char) (i + 'a');
            getListDFS(child, word);
        }
    }

}
